cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32s3)
set(EXTRA_COMPONENT_DIRS "../common" "../common/runtime" "../common/connectivity" "../common/logger" "../common/location" "../common/sensors" "../common/power")
set(AQ_VARIANT logger CACHE STRING "Firmware variant: logger or diagnostic")
if(NOT AQ_VARIANT MATCHES "^(logger|diagnostic)$")
    message(FATAL_ERROR "AQ_VARIANT must be logger or diagnostic")
endif()
set(COMPONENTS aq_waveshare_${AQ_VARIANT})
list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/components/aq_waveshare_${AQ_VARIANT}")
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../../tools/idf-sdkconfig.defaults")
if(AQ_VARIANT STREQUAL "logger")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../../tools/idf-connectivity-sdkconfig.defaults")
endif()
list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
if(AQ_VARIANT STREQUAL "diagnostic")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/diagnostic/sdkconfig.defaults")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/aq_cxx_standard.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/aq_source_paths.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/aq_nimble_patches.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/aq_stack_usage.cmake")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
aq_enable_stack_usage()
aq_map_debug_paths()
# The diagnostic variant pulls different component manifests, so it keeps its
# own lock and building it never rewrites the logger's manifest_hash.
if(AQ_VARIANT STREQUAL "diagnostic")
    idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_CURRENT_LIST_DIR}/diagnostic/dependencies.lock")
endif()
project(aq_waveshare)
aq_patch_nimble()
