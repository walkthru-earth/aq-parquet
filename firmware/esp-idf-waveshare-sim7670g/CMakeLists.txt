cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32s3)
set(EXTRA_COMPONENT_DIRS "../common" "../common/runtime" "../common/connectivity" "../common/logger" "../common/location" "../common/sensors" "../common/power")
set(OS_VARIANT logger CACHE STRING "Firmware variant: logger or diagnostic")
if(NOT OS_VARIANT MATCHES "^(logger|diagnostic)$")
    message(FATAL_ERROR "OS_VARIANT must be logger or diagnostic")
endif()
set(COMPONENTS os_waveshare_${OS_VARIANT})
list(APPEND EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/components/os_waveshare_${OS_VARIANT}")
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../../tools/idf-sdkconfig.defaults")
if(OS_VARIANT STREQUAL "logger")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../../tools/idf-connectivity-sdkconfig.defaults")
endif()
list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
if(OS_VARIANT STREQUAL "diagnostic")
    list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/diagnostic/sdkconfig.defaults")
endif()
# Bench images only. tools/build-idf-board.sh sets it from OS_BENCH and builds
# in a separate directory, so a shipped build never picks it up.
set(OS_SDKCONFIG_OVERLAY "" CACHE FILEPATH "Bench sdkconfig defaults appended last")
if(OS_SDKCONFIG_OVERLAY)
    list(APPEND SDKCONFIG_DEFAULTS "${OS_SDKCONFIG_OVERLAY}")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/os_cxx_standard.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/os_source_paths.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/os_nimble_patches.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/os_stack_usage.cmake")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
os_enable_stack_usage()
os_map_debug_paths()
# The diagnostic variant pulls different component manifests, so it keeps its
# own lock and building it never rewrites the logger's manifest_hash.
if(OS_VARIANT STREQUAL "diagnostic")
    idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_CURRENT_LIST_DIR}/diagnostic/dependencies.lock")
endif()
project(opensensor_waveshare)
os_patch_nimble()
